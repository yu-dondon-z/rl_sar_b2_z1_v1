option(VA15_READ_ONLY_DDS "Build subscriber-only hardware probe; never a controller" OFF)
option(VA15_DIAGNOSTIC_SANITIZERS "Instrument read-only diagnostics with ASan/UBSan" OFF)
function(independent_sanitize target)
  if(VA15_DIAGNOSTIC_SANITIZERS)
    target_compile_options(${target} PRIVATE -O1 -g1 -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${target} PRIVATE -fsanitize=address,undefined)
  endif()
endfunction()
independent_sanitize(independent_probe)
if(VA15_READ_ONLY_DDS)
  set(VA15_SDK_ROOT "" CACHE PATH "Existing Unitree SDK2 root")
  if(NOT EXISTS "${VA15_SDK_ROOT}/include/unitree/robot/channel/channel_subscriber.hpp")
    message(FATAL_ERROR "Explicit existing SDK root required")
  endif()
  target_compile_definitions(independent_probe PRIVATE VA15_READ_ONLY_DDS)
  target_include_directories(independent_probe SYSTEM PRIVATE "${VA15_SDK_ROOT}/include"
    "${VA15_SDK_ROOT}/thirdparty/include" "${VA15_SDK_ROOT}/thirdparty/include/ddscxx")
  target_link_libraries(independent_probe PRIVATE "${VA15_SDK_ROOT}/lib/x86_64/libunitree_sdk2.a"
    "${VA15_SDK_ROOT}/thirdparty/lib/x86_64/libddsc.so" "${VA15_SDK_ROOT}/thirdparty/lib/x86_64/libddscxx.so")
  # Unitree's C++ DDS library requires its bundled C DDS library, not a ROS
  # Humble libddsc with the same SONAME. Pin this standalone diagnostic only.
  set_target_properties(independent_probe PROPERTIES BUILD_RPATH "${VA15_SDK_ROOT}/thirdparty/lib/x86_64")
  target_link_options(independent_probe PRIVATE -Wl,--disable-new-dtags)
endif()
