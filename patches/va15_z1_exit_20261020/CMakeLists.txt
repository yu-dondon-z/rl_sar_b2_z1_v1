cmake_minimum_required(VERSION 3.16)
project(va15_z1_exit_regression LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(VA15_TEST_WORK "${CMAKE_CURRENT_SOURCE_DIR}/../va15_stop_feedback_20261020/work" CACHE PATH "Staged shutdown helpers/tests")
set(VA15_Z1_SDK_ROOT "/home/dong/rl_sar_b2_z1/src/rl_sar/library/thirdparty/z1_sdk" CACHE PATH "Existing Z1 SDK headers/library")
find_package(Threads REQUIRED)
find_program(PYTHON_EXECUTABLE python3 REQUIRED)
enable_testing()
foreach(NAME test_va15_z1_exit test_va15_z1_udp_exit)
  add_executable(${NAME} "${VA15_TEST_WORK}/${NAME}.cpp")
  target_compile_options(${NAME} PRIVATE -UNDEBUG)
  target_include_directories(${NAME} SYSTEM PRIVATE "${VA15_Z1_SDK_ROOT}/include")
  target_link_libraries(${NAME} PRIVATE Threads::Threads)
endforeach()
target_link_libraries(test_va15_z1_udp_exit PRIVATE "${VA15_Z1_SDK_ROOT}/lib/libZ1_SDK_x86_64.so")
add_test(NAME passive_exit_mock COMMAND test_va15_z1_exit)
add_test(NAME passive_exit_real_sdk_loopback_only COMMAND test_va15_z1_udp_exit)
add_test(NAME launcher_fake_process_signal_isolation COMMAND "${PYTHON_EXECUTABLE}" "${VA15_TEST_WORK}/test_local_z1_controller.py")
set_tests_properties(passive_exit_mock passive_exit_real_sdk_loopback_only
  launcher_fake_process_signal_isolation PROPERTIES TIMEOUT 20)
