cmake_minimum_required(VERSION 3.16)
project(va15_stop_offline LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(STAGED "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
set(VA15_BASELINE_ROOT "/home/dong/rl_sar_b2_z1" CACHE PATH "Read-only framework/dependency checkout")
set(VA15_TORCH_ROOT "/home/dong/rl_sar_b2_z1_old/library/inference_runtime/libtorch" CACHE PATH "Existing CPU LibTorch")
set(CORE "${VA15_BASELINE_ROOT}/src/rl_sar/library/core")
set(THIRD "${VA15_BASELINE_ROOT}/src/rl_sar/library/thirdparty")
find_package(Torch REQUIRED PATHS "${VA15_TORCH_ROOT}/share/cmake/Torch" NO_DEFAULT_PATH)
find_package(Threads REQUIRED)
find_package(Eigen3 REQUIRED)
add_compile_definitions(USE_CMAKE USE_TORCH POLICY_DIR="${STAGED}")
add_compile_options(-UNDEBUG)
include_directories("${STAGED}")
foreach(DIR observation_buffer rl_sdk loop fsm vector_math motion_loader inference_runtime logger)
  include_directories("${CORE}/${DIR}")
endforeach()
add_library(va15_core STATIC "${STAGED}/rl_sdk.cpp"
  "${CORE}/observation_buffer/observation_buffer.cpp"
  "${CORE}/motion_loader/motion_loader.cpp"
  "${CORE}/inference_runtime/inference_runtime.cpp")
target_link_libraries(va15_core PUBLIC ${TORCH_LIBRARIES} yaml-cpp tbb Threads::Threads Eigen3::Eigen)
add_executable(test_va15_stop_fsm "${STAGED}/test_va15_stop_fsm.cpp")
target_link_libraries(test_va15_stop_fsm PRIVATE va15_core)
add_executable(va15_native_offline "${STAGED}/rl_real_b2_z1_no_gun.cpp")
add_executable(test_va15_kinematics_shadow "${STAGED}/test_va15_kinematics_shadow.cpp")
target_link_libraries(test_va15_kinematics_shadow PRIVATE yaml-cpp Eigen3::Eigen)
add_executable(replay_va15_kinematics_shadow "${STAGED}/replay_va15_kinematics_shadow.cpp")
target_link_libraries(replay_va15_kinematics_shadow PRIVATE yaml-cpp Eigen3::Eigen)
set(SDK "${THIRD}/robot_sdk/unitree/unitree_sdk2")
target_include_directories(va15_native_offline SYSTEM PRIVATE "${SDK}/include"
  "${SDK}/thirdparty/include" "${SDK}/thirdparty/include/ddscxx" "${THIRD}/z1_sdk/include")
target_link_libraries(va15_native_offline PRIVATE va15_core
  "${SDK}/lib/x86_64/libunitree_sdk2.a"
  "${SDK}/thirdparty/lib/x86_64/libddsc.so" "${SDK}/thirdparty/lib/x86_64/libddscxx.so"
  "${THIRD}/z1_sdk/lib/libZ1_SDK_x86_64.so")
enable_testing()
add_test(NAME stop_fsm COMMAND test_va15_stop_fsm)
add_test(NAME native_contract COMMAND va15_native_offline --offline-check)
add_test(NAME kinematics_math_and_boundaries COMMAND test_va15_kinematics_shadow
  "${STAGED}/b2_z1_no_gun/VA15/b2_leg_kinematics.json"
  "${STAGED}/b2_z1_no_gun/VA15/kinematics_golden.json")
foreach(NAME stop_feedback action_state remote_input stand_gains)
  add_executable(test_va15_${NAME} "${STAGED}/test_va15_${NAME}.cpp")
  add_test(NAME ${NAME} COMMAND test_va15_${NAME})
endforeach()
add_executable(test_va15_joinable_loop "${STAGED}/test_va15_joinable_loop.cpp")
target_link_libraries(test_va15_joinable_loop PRIVATE Threads::Threads)
add_test(NAME joinable_real_loop_lifetime COMMAND test_va15_joinable_loop)
add_executable(test_va15_z1_exit "${STAGED}/test_va15_z1_exit.cpp")
target_include_directories(test_va15_z1_exit SYSTEM PRIVATE "${THIRD}/z1_sdk/include")
target_link_libraries(test_va15_z1_exit PRIVATE Threads::Threads)
add_test(NAME z1_bounded_passive_exit COMMAND test_va15_z1_exit)
add_executable(test_va15_z1_udp_exit "${STAGED}/test_va15_z1_udp_exit.cpp")
target_include_directories(test_va15_z1_udp_exit SYSTEM PRIVATE "${THIRD}/z1_sdk/include")
target_link_libraries(test_va15_z1_udp_exit PRIVATE Threads::Threads "${THIRD}/z1_sdk/lib/libZ1_SDK_x86_64.so")
add_test(NAME z1_loopback_udp_exit COMMAND test_va15_z1_udp_exit)
find_program(PYTHON_EXECUTABLE python3 REQUIRED)
add_test(NAME launcher_mock COMMAND "${PYTHON_EXECUTABLE}" "${STAGED}/test_local_z1_controller.py")
add_test(NAME read_only_sampler COMMAND "${PYTHON_EXECUTABLE}" "${STAGED}/test_sample_b2_stop_feedback.py")
set_tests_properties(stop_fsm native_contract PROPERTIES ENVIRONMENT
  "LD_LIBRARY_PATH=${VA15_TORCH_ROOT}/lib:${SDK}/thirdparty/lib/x86_64:${THIRD}/z1_sdk/lib")
