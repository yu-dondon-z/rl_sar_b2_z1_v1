cmake_minimum_required(VERSION 3.16)
project(vd2_real LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
find_package(Eigen3 REQUIRED)
find_package(Threads REQUIRED)
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
add_subdirectory(thirdparty/unitree_sdk2 EXCLUDE_FROM_ALL)
set(Z1_LIB "${CMAKE_CURRENT_SOURCE_DIR}/thirdparty/z1_sdk/lib/libZ1_SDK_${CMAKE_SYSTEM_PROCESSOR}.so")
if(NOT EXISTS "${Z1_LIB}")
  message(FATAL_ERROR "Unsupported Z1 SDK architecture: ${CMAKE_SYSTEM_PROCESSOR}")
endif()
add_library(vd2_io SHARED native/io.cpp)
target_include_directories(vd2_io PRIVATE thirdparty/z1_sdk/include)
target_link_libraries(vd2_io PRIVATE unitree_sdk2 Eigen3::Eigen Threads::Threads "${Z1_LIB}")
target_link_options(vd2_io PRIVATE "-Wl,--disable-new-dtags")
target_compile_options(vd2_io PRIVATE -Wall -Wextra -Wpedantic)
set(CMAKE_BUILD_RPATH_USE_ORIGIN ON)
set_target_properties(vd2_io PROPERTIES
  LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib"
  BUILD_RPATH "$ORIGIN/../../thirdparty/z1_sdk/lib;$ORIGIN/../../thirdparty/unitree_sdk2/thirdparty/lib/${CMAKE_SYSTEM_PROCESSOR}")
add_executable(vd2_native_check native/check.cpp)
target_link_libraries(vd2_native_check PRIVATE vd2_io)
target_link_options(vd2_native_check PRIVATE "-Wl,--disable-new-dtags")
set_target_properties(vd2_native_check PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin"
  BUILD_RPATH "$ORIGIN/../lib;$ORIGIN/../../thirdparty/z1_sdk/lib;$ORIGIN/../../thirdparty/unitree_sdk2/thirdparty/lib/${CMAKE_SYSTEM_PROCESSOR}")
enable_testing()
add_test(NAME native_contract COMMAND vd2_native_check)
add_executable(vd2_z1_sdk_check native/z1_sdk_check.cpp)
target_include_directories(vd2_z1_sdk_check PRIVATE thirdparty/z1_sdk/include)
target_link_libraries(vd2_z1_sdk_check PRIVATE Eigen3::Eigen Threads::Threads "${Z1_LIB}")
set_target_properties(vd2_z1_sdk_check PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
add_test(NAME z1_sdk_serialization COMMAND vd2_z1_sdk_check)
add_executable(vd2_z1_pacing_check native/z1_pacing_check.cpp)
target_include_directories(vd2_z1_pacing_check PRIVATE thirdparty/z1_sdk/include)
target_link_libraries(vd2_z1_pacing_check PRIVATE Eigen3::Eigen Threads::Threads "${Z1_LIB}")
set_target_properties(vd2_z1_pacing_check PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
add_test(NAME z1_reply_pacing COMMAND vd2_z1_pacing_check)
